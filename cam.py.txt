    def eye_aspect_ratio(self, eye_points):
        """Calculate Eye Aspect Ratio (EAR)"""
        A = np.linalg.norm(eye_points[1] - eye_points[5])
        B = np.linalg.norm(eye_points[2] - eye_points[4])
        C = np.linalg.norm(eye_points[0] - eye_points[3])
        return (A + B) / (2.0 * C)
    
    def levenshtein_distance(self, s1, s2):
        """Calculate Levenshtein distance for spell checking"""
        if len(s1) < len(s2):
            return self.levenshtein_distance(s2, s1)
        if len(s2) == 0:
            return len(s1)
        
        previous_row = range(len(s2) + 1)
        for i, c1 in enumerate(s1):
            current_row = [i + 1]
            for j, c2 in enumerate(s2):
                insertions = previous_row[j + 1] + 1
                deletions = current_row[j] + 1
                substitutions = previous_row[j] + (c1 != c2)
                current_row.append(min(insertions, deletions, substitutions))
            previous_row = current_row
        
        return previous_row[-1]
    
    def auto_correct_word(self, word):
        """Auto-correct word using dictionary"""
        if not word or len(word) < 3:
            return word
        
        upper_word = word.upper()
        if upper_word in self.COMMON_WORDS:
            return word
        
        closest_word = word
        min_distance = float('inf')
        
        for dict_word in self.COMMON_WORDS:
            distance = self.levenshtein_distance(upper_word, dict_word)
            length_diff = abs(len(word) - len(dict_word))
            
            if distance == 1 and length_diff <= 1 and distance < min_distance:
                min_distance = distance
                closest_word = dict_word
        
        if min_distance == 1:
            print(f"✓ Auto-correct: {word} → {closest_word}")
            return closest_word
        
        return word
    
    def decode_morse_buffer(self):
        """Decode morse buffer to characters"""
        chars = self.morse_buffer.strip().split(" ")
        text = ""
        for c in chars:
            if c and c in self.MORSE_TABLE:
                text += self.MORSE_TABLE[c]
        self.current_word = text
        
        # Check for SOS
        morse_sequence = self.morse_buffer.strip().replace("  ", " ")
        if morse_sequence == "... --- ...":
            emergency_msg = "Help me! I need help! I am feeling uneasy!"
            self.decoded_text = self.decoded_text.strip()
            if self.decoded_text:
                self.decoded_text += " " + emergency_msg
            else:
                self.decoded_text = emergency_msg
            self.current_word = ""
            self.morse_buffer = ""
            self.update_display()
            self.speak_text(emergency_msg)
            print("🚨 SOS EMERGENCY DETECTED!")
            messagebox.showwarning("EMERGENCY", "SOS signal detected!")
            return
        
        self.update_display()
    
    def update_display(self):
        """Update text displays"""
        self.morse_text.config(state=tk.NORMAL)
        self.morse_text.delete(1.0, tk.END)
        self.morse_text.insert(1.0, self.morse_buffer)
        self.morse_text.config(state=tk.DISABLED)
        
        full_text = self.decoded_text
        if self.decoded_text and self.current_word:
            full_text += " "
        full_text += self.current_word
        
        self.decoded_text_widget.config(state=tk.NORMAL)
        self.decoded_text_widget.delete(1.0, tk.END)
        self.decoded_text_widget.insert(1.0, full_text or "Your message will appear here...")
        self.decoded_text_widget.config(state=tk.DISABLED)
    
    def speak_text(self, text):
        """Speak text using TTS"""
        if text and text.strip():
            def speak():
                try:
                    self.tts_engine.say(text.strip())
                    self.tts_engine.runAndWait()
                except Exception as e:
                    print(f"TTS Error: {e}")
            threading.Thread(target=speak, daemon=True).start()
    
    def start_camera(self):
        """Start camera and detection"""
        self.cap = cv2.VideoCapture(0)
        if not self.cap.isOpened():
            messagebox.showerror("Error", "Cannot access camera!")
            return
        
        self.camera_active = True
        self.start_btn.config(state=tk.DISABLED)
        self.stop_btn.config(state=tk.NORMAL)
        self.status_label.config(text="● Active", fg='#16a34a')
        
        threading.Thread(target=self.process_camera, daemon=True).start()
        self.save_to_history("📹 Camera Started")
    
    def stop_camera(self):
        """Stop camera"""
        self.camera_active = False
        if self.cap:
            self.cap.release()
        self.start_btn.config(state=tk.NORMAL)
        self.stop_btn.config(state=tk.DISABLED)
        self.status_label.config(text="● Stopped", fg='#ef4444')
        self.canvas.delete("all")
    
    def process_camera(self):
        """Main camera processing loop"""
        while self.camera_active:
            ret, frame = self.cap.read()
            if not ret:
                break
            
            frame = cv2.flip(frame, 1)
            rgb_frame = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
            results = self.face_mesh.process(rgb_frame)
            
            if results.multi_face_landmarks:
                landmarks = results.multi_face_landmarks[0]
                h, w = frame.shape[:2]
                
                left_eye = np.array([(landmarks.landmark[i].x * w, 
                                     landmarks.landmark[i].y * h) 
                                    for i in self.LEFT_EYE])
                right_eye = np.array([(landmarks.landmark[i].x * w, 
                                      landmarks.landmark[i].y * h) 
                                     for i in self.RIGHT_EYE])
                
                cv2.rectangle(frame, 
                            (int(left_eye[:, 0].min()), int(left_eye[:, 1].min())),
                            (int(left_eye[:, 0].max()), int(left_eye[:, 1].max())),
                            (0, 255, 0), 2)
                cv2.rectangle(frame, 
                            (int(right_eye[:, 0].min()), int(right_eye[:, 1].min())),
                            (int(right_eye[:, 0].max()), int(right_eye[:, 1].max())),
                            (0, 255, 0), 2)
                
                left_ear = self.eye_aspect_ratio(left_eye)
                right_ear = self.eye_aspect_ratio(right_eye)
                ear = (left_ear + right_ear) / 2.0
                
                self.detect_blink(ear, left_ear, right_ear)
                
                status = "Paused" if self.is_detection_paused else "Eyes Open"
                color = (255, 165, 0) if self.is_detection_paused else (0, 255, 0)
                if ear < self.settings['blink_threshold']:
                    status = "Eyes Closed"
                    color = (0, 0, 255)
                
                cv2.putText(frame, status, (10, 30), 
                           cv2.FONT_HERSHEY_SIMPLEX, 1, color, 2)
            
            frame = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
            frame = cv2.resize(frame, (640, 480))
            
            try:
                from PIL import Image, ImageTk
                img = Image.fromarray(frame)
                self.photo = ImageTk.PhotoImage(image=img)
                self.canvas.create_image(0, 0, image=self.photo, anchor=tk.NW)
            except:
                pass
            
            time.sleep(0.03)
    
    def detect_blink(self, ear, left_ear, right_ear):
        """Detect blinks and process morse code"""
        now = time.time()
        threshold = self.settings['blink_threshold']
        
        is_left_closed = left_ear < threshold
        is_right_closed = right_ear < threshold
        is_both_open = left_ear >= threshold and right_ear >= threshold
        
        # Left eye wink
        if is_left_closed and not is_right_closed and not self.is_detection_paused:
            if not self.left_eye_closed_time:
                self.left_eye_closed_time = now
        elif self.left_eye_closed_time and is_both_open:
            duration = (now - self.left_eye_closed_time) * 1000
            self.left_eye_closed_time = None
            
            if 300 <= duration <= 800:
                if now - self.last_wink_time < 2:
                    self.wink_sequence.append('L')
                else:
                    self.wink_sequence = ['L']
                self.last_wink_time = now
                
                if len(self.wink_sequence) >= 4 and ''.join(self.wink_sequence[-4:]) == "LRLR":
                    self.clear_message()
                    self.wink_sequence = []
                    return
                
                self.backspace()
        
        # Right eye wink
        if is_right_closed and not is_left_closed and not self.is_detection_paused:
            if not self.right_eye_closed_time:
                self.right_eye_closed_time = now
        elif self.right_eye_closed_time and is_both_open:
            duration = (now - self.right_eye_closed_time) * 1000
            self.right_eye_closed_time = None
            
            if 300 <= duration <= 800:
                if now - self.last_wink_time < 2:
                    self.wink_sequence.append('R')
                else:
                    self.wink_sequence = ['R']
                self.last_wink_time = now
                
                if len(self.wink_sequence) >= 4 and ''.join(self.wink_sequence[-4:]) == "LRLR":
                    self.clear_message()
                    self.wink_sequence = []
                    return
                
                self.backspace()
        
        # Both eyes blink
        if ear < threshold:
            if not self.eye_closed_time:
                self.eye_closed_time = now
            
            closed_duration = (now - self.eye_closed_time) * 1000
            if closed_duration >= 8000:
                self.is_detection_paused = not self.is_detection_paused
                status = "⏸️ Paused" if self.is_detection_paused else "▶️ Active"
                self.status_label.config(text=status)
                self.eye_closed_time = None
        else:
            if self.eye_closed_time and not self.is_detection_paused:
                duration = (now - self.eye_closed_time) * 1000
                self.eye_closed_time = None
                
                self.blink_count += 1
                self.blink_label.config(text=f"Blinks: {self.blink_count}")
                
                if 2000 <= duration < 8000:
                    if self.current_word:
                        self.current_word = ""
                        self.morse_buffer = ""
                    elif self.decoded_text.strip():
                        words = self.decoded_text.strip().split()
                        if words:
                            words.pop()
                            self.decoded_text = " ".join(words) + " " if words else ""
                    self.update_display()
                
                elif duration < self.settings['dot_duration']:
                    self.morse_buffer += "."
                    self.last_blink_time = now
                    self.word_completed = False
                    self.last_sentence_speak = now
                
                elif duration < 2000:
                    self.morse_buffer += "-"
                    self.last_blink_time = now
                    self.word_completed = False
                    self.last_sentence_speak = now
        
        # Character decoding
        if not self.is_detection_paused and self.morse_buffer and \
           not self.morse_buffer.endswith(" ") and \
           (now - self.last_blink_time) > (self.settings['character_pause'] / 1000) and \
           ear >= threshold:
            self.morse_buffer += " "
            self.decode_morse_buffer()
            self.word_completed = False
        
        # Word completion
        time_since_blink = now - self.last_blink_time
        if not self.is_detection_paused and self.current_word and \
           not self.word_completed and \
           time_since_blink > (self.settings['word_pause'] / 1000) and \
           ear >= threshold:
            
            self.word_completed = True
            corrected_word = self.auto_correct_word(self.current_word)
            
            self.decoded_text = self.decoded_text.strip()
            if self.decoded_text:
                self.decoded_text += " " + corrected_word
            else:
                self.decoded_text = corrected_word
            
            self.speak_text(corrected_word)
            self.save_to_history(corrected_word)
            
            self.current_word = ""
            self.morse_buffer = ""
            self.update_display()
            
            self.last_blink_time = now
            self.last_sentence_speak = now
        
        # Sentence repeat
        time_since_activity = now - self.last_sentence_speak
        if not self.is_detection_paused and self.decoded_text.strip() and \
           time_since_activity > (self.settings['sentence_pause'] / 1000) and \
           not self.current_word and ear >= threshold:
            self.speak_text(self.decoded_text.strip())
            self.last_sentence_speak = now
    
    def speak_message(self):
        """Speak current message"""
        full_text = (self.decoded_text + self.current_word).strip()
        if full_text:
            self.speak_text(full_text)
            self.save_to_history(full_text)
    
    def save_message(self):
        """Save message to file"""
        full_text = (self.decoded_text + self.current_word).strip()
        if not full_text:
            messagebox.showwarning("Warning", "No message to save!")
            return
        
        try:
            saved = []
            if os.path.exists(self.saved_messages_file):
                with open(self.saved_messages_file, 'r') as f:
                    saved = json.load(f)
            
            saved.append({
                'text': full_text,
                'timestamp': datetime.now().isoformat(),
                'date': datetime.now().strftime('%Y-%m-%d'),
                'time': datetime.now().strftime('%H:%M:%S')
            })
            
            with open(self.saved_messages_file, 'w') as f:
                json.dump(saved, f, indent=2)
            
            self.save_to_history(full_text)
            messagebox.showinfo("Success", "Message saved successfully!")
        except Exception as e:
            messagebox.showerror("Error", f"Failed to save: {str(e)}")
    
    def clear_message(self):
        """Clear all messages"""
        self.decoded_text = ""
        self.current_word = ""
        self.morse_buffer = ""
        self.blink_count = 0
        self.blink_label.config(text="Blinks: 0")
        self.wink_sequence = []
        self.update_display()
    
    def backspace(self):
        """Delete last character"""
        if self.current_word:
            self.current_word = self.current_word[:-1]
            morse_chars = [c for c in self.morse_buffer.strip().split(" ") if c]
            if morse_chars:
                morse_chars.pop()
                self.morse_buffer = " ".join(morse_chars) + (" " if morse_chars else "")
        elif self.decoded_text:
            self.decoded_text = self.decoded_text[:-1]
        self.update_display()
    
    def add_quick_phrase(self, phrase):
        """Add quick phrase"""
        if self.decoded_text and not self.decoded_text.endswith(" "):
            self.decoded_text += " "
        self.decoded_text += phrase + " "
        self.update_display()
        self.speak_text(phrase)
        self.save_to_history(phrase)
    
    def save_to_history(self, message):
        """Save to history"""
        try:
            history = []
            if os.path.exists(self.history_file):
                with open(self.history_file, 'r') as f:
                    history = json.load(f)
            
            history.insert(0, {
                'text': message,
                'timestamp': time.time()
            })
            
            history = history[:50]
            
            with open(self.history_file, 'w') as f:
                json.dump(history, f, indent=2)
        except Exception as e:
            print(f"Error saving to history: {e}")
    
    def __del__(self):
        """Cleanup on exit"""
        if self.cap:
            self.cap.release()
        cv2.destroyAllWindows()


def main():
    """Main entry point"""
    root = tk.Tk()
    app = Blink2SpeakApp(root)
    
    def on_closing():
        """Handle window close"""
        app.stop_camera()
        root.destroy()
    
    root.protocol("WM_DELETE_WINDOW", on_closing)
    root.mainloop()


if __name__ == "__main__":
    main()
